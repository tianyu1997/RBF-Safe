cmake_minimum_required(VERSION 3.23)
project(RBFSafeConsumer LANGUAGES CXX)
find_package(RBFSafe 5.0 CONFIG REQUIRED)
foreach(required_target core envelope geometry lect atlas applications assurance)
    if(NOT TARGET RBFSafe::${required_target})
        message(FATAL_ERROR "RBFSafe::${required_target} was not exported")
    endif()
endforeach()
add_executable(rbfsafe_consumer main.cpp)
target_link_libraries(rbfsafe_consumer PRIVATE RBFSafe::core RBFSafe::envelope RBFSafe::applications RBFSafe::assurance RBFSafe::rbfsafe RBFSafe::corridor RBFSafe::ik RBFSafe::update RBFSafe::regions RBFSafe::planning RBFSafe::optimization RBFSafe::shield RBFSafe::policy RBFSafe::memory RBFSafe::trust RBFSafe::remote RBFSafe::identity RBFSafe::coordination RBFSafe::deployment RBFSafe::execution RBFSafe::transparency RBFSafe::witness RBFSafe::provenance RBFSafe::occupancy)
target_compile_features(rbfsafe_consumer PRIVATE cxx_std_20)
